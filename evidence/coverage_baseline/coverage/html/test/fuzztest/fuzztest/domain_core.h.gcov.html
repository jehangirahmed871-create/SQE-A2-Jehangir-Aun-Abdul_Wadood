     730                 :             : // ReversibleMap(mapper, inv_mapper, inner...) combinator creates a domain that
     731                 :             : // uses the `mapper` function to map the values created by the `inner...`

     741                 :             : //

     743                 :             : //       [](double real, double imag) {
     744                 :             : //         return std::complex<double>{real, imag};
     745                 :             : //       },
     746                 :             : //       [](std::complex<double> z) {
     747                 :             : //         return std::optional{std::tuple{z.real(), z.imag()}};
     748                 :             : //       },
     749                 :             : //       Arbitrary<double>(), Arbitrary<double>())

     751                 :             : // The function `mapper` doesn't necessarily need to be one-to-one. If it isn't
     752                 :             : // and it maps several inner domain tuples to the same value `y`, then
     753                 :             : // `inv_mapper` can map `y` back to any of these inner domain tuples.
     754                 :             : //
     755                 :             : // Example:
     756                 :             : //
     757                 :             : //   ReversibleMap([](int a, int b) { return std::max(a, b); },
     758                 :             : //                 [](int c) {
     759                 :             : //                   return std::optional{std::tuple{c, c}};
     760                 :             : //                 },
     761                 :             : //                 Arbitrary<int>(), Arbitrary<int>())
     762                 :             : //
     763                 :             : // Importantly, if `inv_mapper` maps `y` to `std::tuple{x...}` and all values
     764                 :             : // `x...` are in the respective inner domains, then `mapper` must map `x...` to
     765                 :             : // `y`. By slightly abusing the notation, we can write this as
     766                 :             : // `mapper(inv_mapper(y)) == y`.

     779                 :             : //       },

     781                 :             : //
     782                 :             : // In this example, `mapper` always returns a pair `(a, b)` such that `a >= b`.

     784                 :             : // no possible value it could return so that
